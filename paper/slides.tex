% --- 8. What broke ---
\begin{frame}{What broke: na\"ive DPO collapses the model}
\begin{itemize}
\item Standard DPO $\Rightarrow$ perception accuracy \textbf{0.95 $\to$ 0.1}; converges to a near-constant ``no''.
\end{itemize}
\vfill
\begin{center}
\setbeamercolor{block body}{bg=red!8}
\begin{block}{The trap}
Answering ``no'' to everything looks \textbf{perfect} on a hallucination metric --- you never affirm anything
false. \textbf{Wrong training metric!}
\end{block}
\end{center}

% --- 9. Read the loss ---
\begin{frame}{Read the loss: what's trained, what's frozen}
\centering
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}r@{\quad}l@{}}
\textcolor{trn}{\textbf{trained}}        & $\theta$ = the two VIB adapters \ (inside: $\mu(x),\ \sigma(x)$) \\
\textcolor{trn}{\textbf{move with }$\theta$} & $c_p,\ r_p$ \ (policy log-probs), \quad $p_\theta$ \\
\textcolor{frz}{\textbf{frozen base}}     & $c_r,\ r_r$ \ (reference $=$ bypass), \quad $p_{\text{base}}$ \\
\textbf{fixed}                            & $\beta,\ \beta_{kl},\ \lambda_a,\ \lambda_{kl},\ \delta$ \ (hyperparams); \ the sigmoid in $-\log\sigma$ \\
\end{tabular}
\vfill
{\small Subscript $p$ = \textcolor{trn}{policy} (adapter \emph{on}) \ $\cdot$ \ subscript $r$ = \textcolor{frz}{reference} (adapter \emph{bypassed} $=$ frozen base).\\
Only $\theta$ is a learnable parameter --- the \textcolor{trn}{policy side} moves, the \textcolor{frz}{reference side} is constant.}
\end{frame}

% --- 10. Why it broke ---
\begin{frame}{Why it broke}
\[ \mathcal L_0=\underbrace{-\log\sigma\!\big(\beta[(c_p{-}c_r){-}(r_p{-}r_r)]\big)}_{\text{swap-DPO margin}}\;+\;\underbrace{\beta_{kl}\,\mathrm{KL}_{\text{VIB}}}_{\text{rate on the adapter only}} \]
\vspace{-0.4em}
\begin{itemize}\small
\item $c_p,r_p$ = policy log-probs of chosen/rejected letters; $c_r,r_r$ = the frozen base.
\item \textbf{Likelihood displacement}: the margin grows while \emph{both} $c_p$ and $r_p$ \emph{fall} $\Rightarrow$ the chosen answer gets \emph{less} likely.
\item $\mathrm{KL}_{\text{VIB}}$ limits the adapter's \emph{rate}, \textbf{not the output} $\Rightarrow$ nothing stops the drift to constant ``no''.
\end{itemize}
\end{frame}

% --- 11. The fix ---
\begin{frame}{The fix: two anchors}
{\small Add two terms to the na\"ive loss:}
\vspace{-0.3em}
\begin{align*}
\mathcal L =\ & \underbrace{-\log\sigma(\beta[(c_p{-}c_r){-}(r_p{-}r_r)])}_{\text{swap-DPO}}
 \;+\; \lambda_a\,\underbrace{\big({-}\log\sigma(\beta(c_p{-}c_r){-}\delta)\big)}_{\textcolor{trn}{\textbf{chosen anchor}}} \\[2pt]
 &+\; \lambda_{kl}\,\underbrace{\mathrm{KL}\!\big(p_{\text{base}}(\cdot|x)\,\|\,p_\theta(\cdot|x)\big)}_{\textcolor{trn}{\textbf{KL-to-base}}}
 \;+\; \beta_{kl}\,\mathrm{KL}_{\text{VIB}}
\end{align*}
\vspace{-0.5em}
\begin{itemize}\small
\item \textcolor{trn}{\textbf{chosen anchor}}: pins the chosen letter's log-prob $\ge$ frozen base $\Rightarrow$ protects \textbf{PA}.
\item \textcolor{trn}{\textbf{KL-to-base}}: on ordinary (non-swap) inputs, keeps the output near base $\Rightarrow$ protects \textbf{HR}.
\end{itemize}
\end{frame}

\begin{frame}{What the metrics mean}
\begin{itemize}
\item \textbf{AVHBench} = the \textbf{target} (audio--visual hallucination); three yes/no tasks:
\vspace{0.06in}
  \begin{itemize}\footnotesize
  \item \textbf{A$\to$V} (audio-driven \emph{video} hall.): ``is the queried object \emph{visible}?'' --- audio tempts a false ``yes''.
  \item \textbf{V$\to$A} (video-driven \emph{audio} hall.): ``is the queried sound \emph{audible}?'' --- \textbf{our audio-grounding probe}.
  \item \textbf{AV-match}: ``do the audio and video \emph{match}?''
  \end{itemize}
\vspace{0.12in}
\item \textbf{CMM} = capability, two axes:
\vspace{0.06in}
  \begin{itemize}\footnotesize
  \item \textbf{PA} (perception): on \emph{present} items --- correctly says ``yes, it's there'' (catches under-perception).
  \item \textbf{HR} (resistance): on \emph{absent} items --- correctly says ``no, it isn't'' (catches hallucination).
  \end{itemize}
\vspace{0.12in}
\item \textbf{DAVE} = \textbf{diagnostic}: audio--visual \textbf{multiple-choice}; each question needs \emph{both} modalities --- a general AV check, \emph{not} a hallucination probe.
\end{itemize}
\end{frame}

% --- 15. What the module actually learned (weights-corrected) ---
\begin{frame}{What the module actually learned (from its weights)}
\begin{itemize}\small
\item \textbf{A small, diffuse re-projection of \emph{both} streams.}
$\lVert W_{\text{out}}\rVert\!\approx\!1$ vs $\lVert W_{\text{enc}}\rVert\!\approx\!26$ (barely off zero-init),
spread over $\sim$570/2048 directions (near-flat spectrum), \textbf{input-driven}
($\lVert\text{bias}\rVert\!\approx\!0.03$).
\item \textbf{Audio and vision adapters are the same magnitude}
($\lVert W_{\text{out}}^{a}\rVert/\lVert W_{\text{out}}^{v}\rVert\!\approx\!1.0$).
\item \textbf{The ``bottleneck'' barely compresses} (KL rate $\sim$0.002): functionally a
\textbf{stochastic zero-init residual re-projection} --- the VIB form buys exploration noise $+$
identity-at-init, not rate-limiting.
\item \textbf{FiLM:} the gate stayed $\approx$open (no hard routing); the learned conditioning is the
\emph{vision-weighted} FiLM scale/shift ($\lVert film_v\rVert\!\approx\!4.5\times\lVert film_a\rVert$).
\end{itemize}
\vfill
{\footnotesize Honest reading: grounding emerges from a \textbf{subtle, roughly symmetric joint
re-encoding} resolved by the frozen LM --- this is exactly what motivates the prompt-aware variant.}
\end{frame}

% --- 16. GRPO ---
\begin{frame}{GRPO on a Frozen AV-LLM via a VIB Adapter}
\textbf{Setup.} The audio-visual LLM is \emph{frozen}; the only trainable
parameters are a tiny \emph{per-modality} Variational Information Bottleneck
(audio, vision) wrapping the modality-adapter outputs.
\[
  z = \mu(x) + \sigma(x)\odot\epsilon,\qquad
  y = x + W_{\text{out}}\,z,\qquad W_{\text{out}}\!\stackrel{\text{init}}{=}0
\]
$W_{\text{out}}$ zero-init $\Rightarrow$ identity at init (base for free);
\textbf{exploration comes from the reparam noise} $\sigma\odot\epsilon$.
\medskip
\textbf{Rollout \& reward (verifiable, no reward model).}
For a yes/no grounding question, draw a group of $G$ samples; the action is a
categorical over the candidate answer tokens $\{\text{yes},\text{no},(\text{abstain})\}$.
\[
  r_i = +1\ (\text{correct}),\quad 0\ (\text{abstain}),\quad -1\ (\text{hallucinate})
\]
\textbf{Group-relative advantage \& objective.}
\[
  A_i = \frac{r_i-\mathrm{mean}(r)}{\mathrm{std}(r)+\varepsilon},\qquad
  \mathcal{L} = -\tfrac{1}{G}\!\sum_i A_i \log\pi(a_i\mid x)
  \;+\;\beta_{\mathrm{kl}}\,\mathrm{KL}\!\big(q(z|x)\,\|\,\mathcal N(0,I)\big)
  \;+\;\lambda_{\mathrm{ref}}\,\mathrm{KL}(\text{base}\,\|\,\pi)
\]
\end{frame}

  \draw[->] (add)--node[above]{\tiny $y$}(llm);
  \draw[->] (x.north) to[out=35,in=145] node[above,pos=0.5]{\tiny skip: $y=x+\cdots$} (add.north);
  \node[frzs,below=6mm of ms,fill=gray!10] (prior) {$\mathcal N(0,I)$ prior};
  \draw[<->,dashed,thick,trn!80!black] (ms.south)--node[right]{\tiny $\mathrm{KL}_{\text{VIB}}$ (rate)}(prior.north);
  \node[font=\tiny,below=1mm of prior]
    {\textcolor{trn}{$\blacksquare$ trained}\quad\textcolor{frz}{$\blacksquare$ frozen}\quad
     one such branch per modality ($a$, $v$), inserted at the adapter output};
\end{tikzpicture}
\vspace{0.4em}
\begin{center}
\setbeamercolor{block body}{bg=blue!6}
\begin{block}{}
\centering
$\displaystyle I(Z;X)\;\le\;\mathbb E_x\,\mathrm{KL}\!\big(q_\phi(z\mid x)\,\big\|\,\mathcal N(0,I)\big)\;=\;\mathrm{KL}_{\text{VIB}}$
\quad{\scriptsize [Alemi et al., \emph{Deep Variational Information Bottleneck}, ICLR 2017]}
\end{block}
\end{center}
{\scriptsize\textbf{Notation:} $x$ = the modality's token stream (audio or vision adapter output);
$W_e$ = the adapter's encoder weights and $h=\mathrm{GELU}(W_e x)$ its hidden activations;
$\mu(h),\sigma(h)$ = mean/std heads defining the posterior $q_\phi(z\,|\,x)=\mathcal N(\mu,\operatorname{diag}\sigma^2)$
over the stochastic code $z$, with $\phi$ = all adapter parameters and $\epsilon\sim\mathcal N(0,I)$ the
reparametrization noise; $W_{\text{out}}$ = zero-init output map producing the edit added back onto $x$.
The bound says: $\mathrm{KL}_{\text{VIB}}$ caps the \emph{bits $z$ keeps about $x$} (the rate).}
\vfill
\begin{center}
\setbeamercolor{block body}{bg=red!6}
\begin{block}{}
\centering\small \textbf{Every box above is question-blind}: $\mu,\sigma$, the code, the edit are
\emph{identical} for ``what do you \textbf{hear}?'' and ``what do you \textbf{see}?'' ---
the adapter cannot treat them differently \emph{even in principle}.
\end{block}
\end{center}
\end{frame}

% --- 18.ii FiLM math (build): condition it, in three small steps ---
\begin{frame}{FiLM, step by step: give that machinery the question}
\textbf{Step 1 --- the question as one vector} (no new encoder):
\[
q \;=\; W_q\,\underbrace{\tfrac1L\textstyle\sum_{t} E(w_t)}_{\text{frozen embedding, mean-pool}}
\]
\textbf{Step 2 --- let $q$ modulate the hidden activations} (FiLM: a per-feature scale $+$ shift):
\[
[\gamma;\beta]=W_{\!f}\,q \qquad
h\ \longrightarrow\ \tilde h=(1+\mathrm{clamp}(\gamma))\odot h+\beta
\qquad\Rightarrow\qquad \mu(x)\to\mu(x,q),\ \ \sigma(x)\to\sigma(x,q)
\]
\textbf{Step 3 --- let $q$ gate the edit} (one scalar per modality):
\[
g(q)=\mathrm{sigm}(w_g^{\!\top}q+b_g)\in(0,1) \qquad\qquad
\boxed{\;y \;=\; x \;+\; g(q)\,W_{\text{out}}\,z\;}
\]
\vspace{-0.4em}
\begin{itemize}\small
\item \textbf{Proposition (identity at init, for every $q$).} $W_{\text{out}}{=}0,\ W_{\!f}{=}0$
($\gamma{=}\beta{=}0$) $\Rightarrow y\equiv x$ and the KL rate equals the unconditional VIB's.
\emph{Proof:} the residual is $g\cdot 0\cdot z=0$ for any $q,g$. $\square$\quad
So \texttt{bypass} still returns the \textbf{exact frozen base}: the DPO reference stays free.
\item New trainables: $W_{\!f}, w_g$ per modality $+$ one shared $W_q$; everything else as before.
\end{itemize}
\end{frame}

% --- 18.iv FiLM math (the forcing lemma) ---
\begin{frame}{FiLM, formally: why the question \emph{must} be read}
\textbf{Setup.} One swapped clip $x^\star$ (\emph{seen} $A$, \emph{heard} $B$), asked two questions:
\[
q_{\text{hear}}:\ \text{chosen}=B,\ \text{rejected}=A
\qquad\qquad
q_{\text{see}}:\ \text{chosen}=A,\ \text{rejected}=B
\]
\begin{center}
\setbeamercolor{block body}{bg=blue!6}
\begin{block}{Lemma (question-blind adapters have a loss floor)}
\small For any adapter that ignores $q$, the two swap-DPO margins on $x^\star$ are exact negatives,
$m_{\text{see}}=-m_{\text{hear}}$, hence its paired loss obeys
\[
-\log\sigma(m)-\log\sigma(-m)\;=\;-\log\!\big[\sigma(m)\,\sigma(-m)\big]\;\ge\;2\log 2\;\approx\;1.39,
\]
with equality only at $m=0$ (learn \emph{nothing}). A question-\emph{aware} adapter can drive both
losses $\to 0$.
\end{block}
\end{center}
\textbf{Corollary.} Any minimizer of the HEAR$+$SEE objective must condition on $q$:
\textbf{routing is forced by data construction} --- with a quantified gap ($2\log 2$) --- not
encouraged by a regularizer.
\vfill
{\footnotesize The minimizer we actually find (from the trained weights): conditioning lands in the
\emph{vision} FiLM ($\lVert W_{\!f}^{(v)}\rVert\!\approx\!4.5\,\lVert W_{\!f}^{(a)}\rVert$); the gate stays
$\approx$\,open --- question-\emph{modulated} re-encoding rather than hard routing.}
\end{frame}

% --- 28. AV-attention: methodology ---
\begin{frame}{How we measure ``attention to audio--visual evidence'' (MoD-DPO++ Fig.~6)}
\textbf{One \emph{eager}-attention forward per clip} (prompt $+$ media); no generation needed.
\begin{enumerate}\small
\item \textbf{Mark the AV tokens.} Flag the audio/visual placeholder positions in the input
(\texttt{<|audio\_pad|>}, \texttt{<|image\_pad|>}, \texttt{<|video\_pad|>}) $\to$ the set $\mathrm{AV}$;
everything else is question text / chat template.
\item \textbf{Take the answer position.} Use the last prompt token $i^\star$ (it predicts the reply);
causal attention $\Rightarrow$ it attends over the whole prompt.
\item \textbf{Per layer $\ell$, head $h$ --- share of its attention spent on AV:}
\[
\mathrm{frac}_{\ell,h}=\frac{\sum_{j\in\mathrm{AV}} A_{\ell,h,\,i^\star\!,\,j}}
{\sum_{j} A_{\ell,h,\,i^\star\!,\,j}}
\qquad(\text{denominator}\approx 1,\ \text{softmax-normalized})
\]
\item \textbf{Average over heads and layers} $\Rightarrow$ one \% per clip; box-plot over $\sim$150 CMM clips.
\end{enumerate}
\end{frame}
